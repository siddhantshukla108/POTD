import java.util.*;

class Solution {
    public int findMaxPathScore(int[][] edges, boolean[] online, long k) {
        int n = online.length;
        
        // Step 1: Build Adjacency List while filtering out offline nodes
        
        List<int[]>[] adj = new ArrayList[n];
        for (int i = 0; i < n; i++) {
            adj[i] = new ArrayList<>();
        }
        
        int[] inDegree = new int[n];
        long maxCost = -1;
        
        for (int[] edge : edges) {
            int u = edge[0];
            int v = edge[1];
            int cost = edge[2];
            
            // Only include edges where both endpoints are online

            if (online[u] && online[v]) {
                adj[u].add(new int[]{v, cost});
                inDegree[v]++;
                maxCost = Math.max(maxCost, cost);
            }
        }
        
        // Step 2: Compute Topological Sort (Kahn's Algorithm)

        int[] topo = new int[n];
        int idx = 0;
        Queue<Integer> q = new LinkedList<>();
        
        for (int i = 0; i < n; i++) {
            if (inDegree[i] == 0) {
                q.offer(i);
            }
        }
        
        while (!q.isEmpty()) {
            int u = q.poll();
            topo[idx++] = u;
            for (int[] edge : adj[u]) {
                int v = edge[0];
                if (--inDegree[v] == 0) {
                    q.offer(v);
                }
            }
        }
        
        // Step 3: Binary Search for the maximum score

        long low = 0;
        long high = maxCost;
        long ans = -1;
        
        while (low <= high) {
            long mid = low + (high - low) / 2;
            
            // Check if there is a valid path using edges >= mid

            if (canReachWithMinScore(mid, n, adj, topo, k)) {
                ans = mid;       
                low = mid + 1;
            } else {
                high = mid - 1;  
            }
        }
        
        return (int) ans;
    }
    
    private boolean canReachWithMinScore(long minScore, int n, List<int[]>[] adj, int[] topo, long k) {
        long[] dist = new long[n];

        // Use a suitably large value to represent infinity to prevent overflow when adding

        long INF = Long.MAX_VALUE / 2;
        Arrays.fill(dist, INF);
        dist[0] = 0;
        
        // Traverse nodes in topological order to find the shortest path

        for (int u : topo) {
            if (dist[u] == INF) continue;
            
            for (int[] edge : adj[u]) {
                int v = edge[0];
                long cost = edge[1];
                
                // Only traverse edges that meet the current minimum score requirement

                if (cost >= minScore) {
                    if (dist[u] + cost < dist[v]) {
                        dist[v] = dist[u] + cost;
                    }
                }
            }
        }
        
        // Check if node n-1 is reachable and within the allowed total cost k
        
        return dist[n - 1] <= k;
    }
}
