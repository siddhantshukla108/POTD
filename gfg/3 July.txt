class Solution {
    public int waysToIncreaseLCSBy1(String s1, String s2) {
        int n1 = s1.length();
        int n2 = s2.length();

        // Forward LCS DP

        int[][] pref = new int[n1 + 1][n2 + 1];
        for (int i = 1; i <= n1; i++) {
            for (int j = 1; j <= n2; j++) {
                if (s1.charAt(i - 1) == s2.charAt(j - 1))
                    pref[i][j] = pref[i - 1][j - 1] + 1;
                else
                    pref[i][j] = Math.max(pref[i - 1][j], pref[i][j - 1]);
            }
        }

        // Backward LCS DP

        int[][] suff = new int[n1 + 1][n2 + 1];
        for (int i = n1 - 1; i >= 0; i--) {
            for (int j = n2 - 1; j >= 0; j--) {
                if (s1.charAt(i) == s2.charAt(j))
                    suff[i][j] = suff[i + 1][j + 1] + 1;
                else
                    suff[i][j] = Math.max(suff[i + 1][j], suff[i][j + 1]);
            }
        }

        int lcs = pref[n1][n2];
        int ans = 0;

        // Try every insertion position

        for (int pos = 0; pos <= n1; pos++) {

            // Try every character

            for (char ch = 'a'; ch <= 'z'; ch++) {
                boolean ok = false;

                for (int j = 0; j < n2; j++) {
                    if (s2.charAt(j) == ch &&
                        pref[pos][j] + 1 + suff[pos][j + 1] == lcs + 1) {
                        ok = true;
                        break;
                    }
                }

                if (ok) ans++;
            }
        }

        return ans;
    }
}
